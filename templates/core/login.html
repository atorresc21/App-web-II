def login(request): if request.method == "POST": correo = request.POST.get("correo", "").strip() clave = request.POST.get("clave", "") Usuario = get_user_model() usuario = Usuario.objects.filter( email__iexact=correo ).first() if usuario: usuario_autenticado = authenticate( request, username=usuario.get_username(), password=clave ) if usuario_autenticado is not None: auth_login(request, usuario_autenticado) return redirect("panel") return render(request, "core/login.html", { "error": "Correo electrónico o contraseña incorrectos." }) return render(request, "core/login.html")
